cmake_minimum_required(VERSION 3.28)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
if(NOT DEFINED ENV{HPM_SDK_BASE})
    set(ENV{HPM_SDK_BASE} "${CMAKE_CURRENT_SOURCE_DIR}/../rmcs_board/bsp/hpm_sdk")
endif()
find_package(hpm-sdk REQUIRED HINTS $ENV{HPM_SDK_BASE})
project(evk_ecat LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS OFF)
include(cmake/soes.cmake)
target_link_libraries(evk_soes PUBLIC ${HPM_SDK_LIB_ITF})
target_link_libraries(app PRIVATE evk_soes)
set_target_properties(app PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_compile_definitions(evk_soes PUBLIC BYTE_ORDER=1234 LITTLE_ENDIAN=1234 BIG_ENDIAN=4321)
sdk_app_inc("${CMAKE_CURRENT_SOURCE_DIR}/../..")
sdk_app_inc("${CMAKE_CURRENT_SOURCE_DIR}/../../core/include")
sdk_app_src("${CMAKE_CURRENT_SOURCE_DIR}/../../core/src/protocol/deserializer.cpp")
sdk_compile_definitions(-DLIBRMCS_DEBUG_ASSERTS=0)
sdk_compile_options($<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>)
sdk_compile_options($<$<COMPILE_LANGUAGE:CXX>:-fno-rtti>)
sdk_compile_options(-O2)
sdk_app_src(app/src/main.cpp app/src/ecat/hardware.cpp app/src/ecat/object_dictionary.cpp)
